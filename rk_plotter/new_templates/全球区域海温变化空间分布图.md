一、图像类型识别与结构拆解

这段代码绘制的是一张**全球/区域海温变化空间分布图 / 经纬度栅格填色地图 / contourf map**。

图像结构为：

```text
单面板空间分布图
投影方式：PlateCarree，central_longitude=180
空间范围：30°E–330°E，60°S–60°N
填色变量：ΔSST，模拟海表温度变化
图层结构：contourf 填色场 + 灰色陆地 + 黑色海岸线 + 经纬网格线 + 水平 colorbar
```

主要视觉特征：

- 海洋区域使用连续色带表示 `ΔSST`。
    
- 使用 `RdYlBu_r` 反向红黄蓝色带，高值偏红，低值偏蓝。
    
- 陆地为浅灰色。
    
- 海岸线为黑色细线。
    
- 经纬线为灰色虚线。
    
- 色标位于图下方，为水平 colorbar。
    
- 标题位于左上角：`A. ΔSST (CMIP6)`。

## 二、单独的虚拟数据生成代码

```python
import numpy as np
import pandas as pd


def generate_virtual_sst_change_data(seed=42):
    """
    生成用于绘制 ΔSST 空间分布图的虚拟经纬度栅格数据。

    返回：
    lons   : 经度，一维数组，范围 0–360
    lats   : 纬度，一维数组，范围 -60–60
    data   : 模拟 ΔSST 数据，二维数组，单位可理解为 °C
    df     : 长表格式数据，便于后续替换为真实数据
    """

    rng = np.random.default_rng(seed)

    lons = np.linspace(0, 360, 180)
    lats = np.linspace(-60, 60, 90)
    lon2d, lat2d = np.meshgrid(lons, lats)

    # 基础纬向增温：低纬度较高，高纬度相对较低
    base = 2.5 + 2.0 * np.cos(np.deg2rad(lat2d))

    # 加入经向波动，模拟东西向海盆差异
    basin_pattern = (
        0.35 * np.sin(np.deg2rad(lon2d * 1.4))
        + 0.25 * np.cos(np.deg2rad(lon2d * 2.2 + lat2d))
    )

    # 加入局地暖异常中心
    warm_pool = (
        0.65 * np.exp(-((lon2d - 150) ** 2 / (2 * 35 ** 2) + (lat2d - 5) ** 2 / (2 * 18 ** 2)))
        + 0.45 * np.exp(-((lon2d - 220) ** 2 / (2 * 42 ** 2) + (lat2d + 10) ** 2 / (2 * 20 ** 2)))
    )

    # 随机扰动
    noise = rng.normal(0, 0.18, size=lon2d.shape)

    data = base + basin_pattern + warm_pool + noise
    data = np.clip(data, 0, 6)

    df = pd.DataFrame({
        "lon": lon2d.ravel(),
        "lat": lat2d.ravel(),
        "delta_sst": data.ravel()
    })

    return lons, lats, data, df


if __name__ == "__main__":
    lons, lats, data, df = generate_virtual_sst_change_data()

    df.to_csv("virtual_delta_sst_data.csv", index=False)

    print(df.head())
```

## 三、完整 Python 绘图代码

```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import matplotlib.ticker as mticker
import cartopy.crs as ccrs
import cartopy.feature as cfeature


# ============================================================
# 1. 虚拟数据生成
# ============================================================

def generate_virtual_sst_change_data(seed=42):
    rng = np.random.default_rng(seed)

    lons = np.linspace(0, 360, 180)
    lats = np.linspace(-60, 60, 90)
    lon2d, lat2d = np.meshgrid(lons, lats)

    base = 2.5 + 2.0 * np.cos(np.deg2rad(lat2d))

    basin_pattern = (
        0.35 * np.sin(np.deg2rad(lon2d * 1.4))
        + 0.25 * np.cos(np.deg2rad(lon2d * 2.2 + lat2d))
    )

    warm_pool = (
        0.65 * np.exp(-((lon2d - 150) ** 2 / (2 * 35 ** 2) + (lat2d - 5) ** 2 / (2 * 18 ** 2)))
        + 0.45 * np.exp(-((lon2d - 220) ** 2 / (2 * 42 ** 2) + (lat2d + 10) ** 2 / (2 * 20 ** 2)))
    )

    noise = rng.normal(0, 0.18, size=lon2d.shape)

    data = base + basin_pattern + warm_pool + noise
    data = np.clip(data, 0, 6)

    return lons, lats, data


lons, lats, data = generate_virtual_sst_change_data()


# ============================================================
# 2. 基础样式与配色
# ============================================================

plt.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Arial", "DejaVu Sans"],
    "axes.linewidth": 1.0,
    "xtick.direction": "out",
    "ytick.direction": "out",
    "savefig.dpi": 300,
})

# 连续色带：Matplotlib 内置 RdYlBu_r
# 低值端为蓝色，高值端为红色
cmap_name = "RdYlBu_r"
levels = np.arange(0, 6.1, 0.5)

land_color = "#d3d3d3"
coastline_color = "#000000"
gridline_color = "#808080"
background_color = "#ffffff"


# ============================================================
# 3. 创建画布与地图投影
# ============================================================

fig = plt.figure(figsize=(10, 6), dpi=130)

ax = plt.subplot(
    1,
    1,
    1,
    projection=ccrs.PlateCarree(central_longitude=180)
)

ax.set_facecolor(background_color)


# ============================================================
# 4. 绘制 ΔSST 填色图
# ============================================================

cf = ax.contourf(
    lons,
    lats,
    data,
    levels=levels,
    transform=ccrs.PlateCarree(),
    cmap=cmap_name,
    extend="both",
    zorder=0
)


# ============================================================
# 5. 添加陆地、海岸线与边界范围
# ============================================================

ax.add_feature(
    cfeature.LAND,
    facecolor=land_color,
    edgecolor="none",
    zorder=2
)

ax.coastlines(
    resolution="110m",
    color=coastline_color,
    linewidth=0.8,
    zorder=3
)

ax.set_extent(
    [30, 330, -60, 60],
    crs=ccrs.PlateCarree()
)


# ============================================================
# 6. 经纬网格与标签
# ============================================================

gl = ax.gridlines(
    crs=ccrs.PlateCarree(),
    draw_labels=True,
    linewidth=0.7,
    linestyle="--",
    color=gridline_color,
    alpha=0.5
)

gl.top_labels = False
gl.right_labels = False

gl.xlocator = mticker.FixedLocator([40, 100, 160, 220, 280, 340])
gl.ylocator = mticker.FixedLocator([-60, -30, 0, 30, 60])

gl.xlabel_style = {
    "size": 10,
    "color": "#000000"
}

gl.ylabel_style = {
    "size": 10,
    "color": "#000000"
}

gl.xformatter = mticker.FixedFormatter([
    "40°E",
    "100°E",
    "160°E",
    "140°W",
    "80°W",
    "20°W"
])

gl.yformatter = mticker.FixedFormatter([
    "60°S",
    "30°S",
    "0°",
    "30°N",
    "60°N"
])


# ============================================================
# 7. 标题与颜色条
# ============================================================

ax.set_title(
    r"A. $\Delta$SST (CMIP6)",
    loc="left",
    fontsize=14,
    fontweight="bold",
    pad=10
)

cb = plt.colorbar(
    cf,
    ax=ax,
    orientation="horizontal",
    pad=0.10,
    shrink=0.62,
    aspect=24
)

cb.set_ticks([0, 2, 4, 6])
cb.ax.tick_params(
    labelsize=10,
    length=4,
    width=0.8,
    direction="out"
)

cb.set_label(
    r"$\Delta$SST (°C)",
    fontsize=11,
    labelpad=6
)

cb.outline.set_linewidth(0.8)


# ============================================================
# 8. 保存与显示
# ============================================================

plt.savefig(
    "reproduced_delta_sst_cmip6_map.png",
    dpi=300,
    bbox_inches="tight",
    pad_inches=0.04
)

plt.show()
```
