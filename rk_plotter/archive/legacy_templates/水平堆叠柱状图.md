一、图像类型识别与结构拆解

这是一个**横向堆叠条形图 / 水平堆叠柱状图**，用于比较不同排名区域 `R1–R10` 的 MP 排放量组成。

图中每个排名区域包含两部分：

- 蓝绿色：`WWTP`
    
- 浅绿色：`Untreated wastewater`
    
- 横轴：`MP discharge (10^12 particles)`
    
- 纵轴：`Ranking`
    
- 条形按总排放量大致从高到低排列
    
- 图例放在图内右上角，文字竖排
    
- 条形带有细密斜线 hatch 纹理
    
- 整体是典型论文图风格：白底、细边框、淡网格、紧凑布局

## 三、完整 Python 绘图代码

```python
import pandas as pd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
matplotlib.rcParams["svg.fonttype"] = "none"
from matplotlib.patches import Rectangle


# =========================
# 1. 虚拟数据
# =========================

def generate_mp_discharge_data():
    data = {
        "Ranking": ["R1", "R2", "R3", "R4", "R5", "R6", "R7", "R8", "R9", "R10"],
        "WWTP": [13.0, 9.6, 11.5, 10.4, 8.3, 9.8, 8.6, 7.8, 10.3, 7.2],
        "Untreated wastewater": [11.7, 14.3, 12.0, 9.7, 10.4, 7.8, 8.4, 8.1, 5.2, 7.8],
    }

    df = pd.DataFrame(data)
    df["Total"] = df["WWTP"] + df["Untreated wastewater"]
    return df


df = generate_mp_discharge_data()


# =========================
# 2. 基础样式
# =========================

plt.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Arial", "DejaVu Sans"],
    "axes.linewidth": 1.0,
    "hatch.linewidth": 0.45,
    "xtick.direction": "out",
    "ytick.direction": "out",
    "savefig.dpi": 300,
})

color_wwtp = "#2f9db3"
color_untreated = "#9ed7b8"
edge_color = "#4f6868"


# =========================
# 3. 绘图
# =========================

fig, ax = plt.subplots(figsize=(5.55, 3.92), dpi=120)

y = range(len(df))
bar_height = 0.62

ax.barh(
    y,
    df["WWTP"],
    height=bar_height,
    color=color_wwtp,
    edgecolor=edge_color,
    linewidth=0.9,
    hatch="////",
    zorder=3,
    label="WWTP"
)

ax.barh(
    y,
    df["Untreated wastewater"],
    left=df["WWTP"],
    height=bar_height,
    color=color_untreated,
    edgecolor=edge_color,
    linewidth=0.9,
    hatch="////",
    zorder=3,
    label="Untreated wastewater"
)


# =========================
# 4. 坐标轴与网格
# =========================

ax.set_yticks(list(y))
ax.set_yticklabels(df["Ranking"], fontsize=15)
ax.invert_yaxis()

ax.set_xlim(0, 30)
ax.set_xticks(range(0, 31, 5))
ax.tick_params(axis="x", labelsize=15, length=0)
ax.tick_params(axis="y", labelsize=15, length=0)

ax.set_xlabel(r"MP discharge ($10^{12}$ particles)", fontsize=17)
ax.set_ylabel("Ranking", fontsize=17)

ax.grid(
    axis="x",
    color="#eeeeee",
    linewidth=0.8,
    zorder=0
)

for spine in ax.spines.values():
    spine.set_visible(True)
    spine.set_color("#808080")
    spine.set_linewidth(1.0)


# =========================
# 5. 仿原图右上角竖排图例
# =========================

legend_y_top = -0.35
legend_h = 1.25
legend_w = 1.25

# Untreated wastewater 图例
ax.add_patch(
    Rectangle(
        (25.7, legend_y_top),
        legend_w,
        legend_h,
        facecolor=color_untreated,
        edgecolor=edge_color,
        linewidth=0.9,
        hatch="////",
        clip_on=False,
        zorder=5
    )
)

ax.text(
    27.15,
    1.15,
    "Untreated wastewater",
    rotation=-90,
    ha="center",
    va="center",
    fontsize=14
)

# WWTP 图例
ax.add_patch(
    Rectangle(
        (27.7, legend_y_top),
        legend_w,
        legend_h,
        facecolor=color_wwtp,
        edgecolor=edge_color,
        linewidth=0.9,
        hatch="////",
        clip_on=False,
        zorder=5
    )
)

ax.text(
    29.15,
    1.15,
    "WWTP",
    rotation=-90,
    ha="center",
    va="center",
    fontsize=14
)


# =========================
# 6. 子图编号
# =========================

ax.text(
    -0.16,
    1.04,
    "b",
    transform=ax.transAxes,
    ha="left",
    va="bottom",
    fontsize=23,
    fontweight="bold"
)


# =========================
# 7. 保存与显示
# =========================

plt.subplots_adjust(
    left=0.18,
    right=0.98,
    bottom=0.20,
    top=0.95
)

plt.savefig(
    "reproduced_mp_discharge_stacked_bar.png",
    dpi=300,
    bbox_inches="tight"
)

plt.close(fig)
```

## 四、可调整参数说明

```python
bar_height = 0.62
```

控制条形粗细。

```python
color_wwtp = "#2f9db3"
color_untreated = "#9ed7b8"
```

控制两组堆叠颜色。

```python
hatch="////"
```

控制条形内部斜线纹理。若想更接近原图，可以使用：

```python
hatch="//////"
```

```python
ax.set_xlim(0, 30)
ax.set_xticks(range(0, 31, 5))
```

控制横轴范围与刻度。
