一、图像类型识别与结构拆解

这是一个**全球空间专题图 / 世界尺度分级着色地图 + 气泡点图**。

图像主体由三层空间信息组成：

```text
世界国家底图 + 国家尺度 MeHg export 分级着色 + 点位尺度 MeHg yield 气泡
```

图中主要元素包括：

- 世界地图底图：浅灰色背景，国家边界为白色线。
    
- 国家填色：蓝色渐变，表示 `MeHg export (kg yr⁻¹)`。
    
- 气泡点：粉红色圆点，大小表示 `MeHg yield (ng m⁻² yr⁻¹)`。
    
- 左侧图例：气泡大小等级。
    
- 底部横向色标：`MeHg export (kg yr⁻¹)`，对数刻度。
    
- 整体风格：典型 Nature / Science 风格全球环境地球化学空间分布图。

## 二、单独的虚拟数据生成代码

建议保存为：`virtual_mehg_worldmap_data.py`

```python
import numpy as np
import pandas as pd


def generate_country_export_data(world, seed=42):
    """
    为 Natural Earth 世界国家边界生成虚拟 MeHg export 数据。
    单位：kg yr^-1
    """
    rng = np.random.default_rng(seed)

    df = world[["ADMIN", "ISO_A3", "CONTINENT"]].copy()

    continent_factor = {
        "Asia": 1.6,
        "Europe": 1.2,
        "Africa": 0.8,
        "North America": 1.4,
        "South America": 1.3,
        "Oceania": 0.6,
        "Seven seas (open ocean)": 0.2,
        "Antarctica": 0.02,
    }

    base = []
    for _, row in df.iterrows():
        factor = continent_factor.get(row["CONTINENT"], 1.0)
        value = 10 ** rng.uniform(-2.0, 3.1) * factor
        base.append(value)

    df["MeHg_export"] = np.clip(base, 0.01, 3000)

    # 强化部分大河流域国家，使视觉更接近原图
    boosts = {
        "Brazil": 1500,
        "United States of America": 1200,
        "China": 2200,
        "India": 950,
        "Indonesia": 800,
        "Russia": 1300,
        "Dem. Rep. Congo": 600,
        "Peru": 500,
        "Colombia": 480,
        "Australia": 350,
    }

    for country, value in boosts.items():
        df.loc[df["ADMIN"] == country, "MeHg_export"] = value

    return df


def generate_mehg_yield_points(seed=7):
    """
    生成虚拟点位数据。
    lon, lat: 经纬度
    MeHg_yield: ng m^-2 yr^-1，用于控制气泡大小
    """
    rng = np.random.default_rng(seed)

    clusters = [
        # name, lon, lat, n, mean_yield
        ("Amazon", -55, -5, 18, 580),
        ("Mississippi", -91, 31, 12, 250),
        ("Central America", -84, 13, 10, 220),
        ("Europe", 10, 48, 28, 180),
        ("West Africa", -5, 8, 12, 170),
        ("Congo", 20, 0, 10, 230),
        ("Nile", 31, 28, 10, 140),
        ("India", 78, 22, 18, 260),
        ("Ganges", 88, 24, 12, 360),
        ("Yangtze", 115, 30, 18, 320),
        ("Southeast Asia", 104, 12, 22, 420),
        ("Indonesia", 115, -4, 28, 520),
        ("Japan_Korea", 137, 36, 10, 180),
        ("Australia", 145, -25, 12, 180),
        ("Pacific Islands", 165, -15, 10, 210),
        ("South Africa", 25, -28, 8, 150),
        ("Russia", 95, 58, 14, 120),
        ("Canada", -105, 56, 16, 120),
    ]

    rows = []

    for name, lon0, lat0, n, mean_yield in clusters:
        lon = rng.normal(lon0, 7.0, n)
        lat = rng.normal(lat0, 4.2, n)

        y = rng.gamma(shape=2.0, scale=mean_yield / 2.0, size=n)
        y = np.clip(y, 20, 950)

        for lo, la, yy in zip(lon, lat, y):
            rows.append({
                "cluster": name,
                "lon": lo,
                "lat": la,
                "MeHg_yield": yy,
            })

    # 增加少量全球零散点
    n_random = 70
    lon = rng.uniform(-175, 178, n_random)
    lat = rng.uniform(-50, 70, n_random)
    y = rng.gamma(shape=1.5, scale=80, size=n_random)
    y = np.clip(y, 10, 420)

    for lo, la, yy in zip(lon, lat, y):
        rows.append({
            "cluster": "Random",
            "lon": lo,
            "lat": la,
            "MeHg_yield": yy,
        })

    points = pd.DataFrame(rows)

    # 限制在图面范围内
    points = points[
        (points["lon"] >= -180) &
        (points["lon"] <= 180) &
        (points["lat"] >= -58) &
        (points["lat"] <= 82)
    ].reset_index(drop=True)

    return points


if __name__ == "__main__":
    print("This script provides:")
    print("1. generate_country_export_data(world)")
    print("2. generate_mehg_yield_points()")
```

## 三、完整 Python 绘图代码

> 运行前需要安装：`geopandas cartopy matplotlib pandas numpy`

```python
import numpy as np
import pandas as pd
import geopandas as gpd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
matplotlib.rcParams["svg.fonttype"] = "none"

from matplotlib.colors import LogNorm, LinearSegmentedColormap
from matplotlib.cm import ScalarMappable
from matplotlib.ticker import FuncFormatter
from cartopy.io import shapereader as shpreader

from virtual_mehg_worldmap_data import (
    generate_country_export_data,
    generate_mehg_yield_points
)


# ============================================================
# 1. 读取世界国家边界
# ============================================================

def load_world_boundaries():
    """
    使用 cartopy 自动获取 Natural Earth 世界国家边界。
    第一次运行时可能会自动下载 Natural Earth 数据。
    """
    shp_path = shpreader.natural_earth(
        resolution="110m",
        category="cultural",
        name="admin_0_countries"
    )

    world = gpd.read_file(shp_path)

    # 统一字段名称，便于不同版本兼容
    if "ADMIN" not in world.columns and "NAME" in world.columns:
        world["ADMIN"] = world["NAME"]

    if "ISO_A3" not in world.columns and "ADM0_A3" in world.columns:
        world["ISO_A3"] = world["ADM0_A3"]

    return world


world = load_world_boundaries()

country_export = generate_country_export_data(world, seed=42)
points = generate_mehg_yield_points(seed=7)

world_plot = world.merge(
    country_export[["ADMIN", "MeHg_export"]],
    on="ADMIN",
    how="left"
)

world_plot["MeHg_export"] = world_plot["MeHg_export"].fillna(0.01)


# ============================================================
# 2. 样式参数
# ============================================================

plt.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Arial", "DejaVu Sans"],
    "axes.linewidth": 1.1,
    "savefig.dpi": 300,
})

blue_cmap = LinearSegmentedColormap.from_list(
    "meh_export_blue",
    [
        "#74aef2",
        "#5d9fe6",
        "#3f86d1",
        "#216db8",
        "#0b55a5",
        "#003f88"
    ],
    N=256
)

norm = LogNorm(vmin=0.01, vmax=3000)


def yield_to_size(y):
    """
    将 MeHg yield 映射为气泡面积。
    """
    return 8 + (np.asarray(y) / 600) ** 1.45 * 260


# ============================================================
# 3. 创建画布
# ============================================================

fig = plt.figure(figsize=(9.4, 4.85), dpi=120)

# 主地图区域
ax = fig.add_axes([0.02, 0.24, 0.96, 0.70])
ax.set_facecolor("#d9d9d9")

ax.set_xlim(-180, 180)
ax.set_ylim(-58, 84)
ax.set_aspect("equal")
ax.set_xticks([])
ax.set_yticks([])

for spine in ax.spines.values():
    spine.set_visible(True)
    spine.set_color("black")
    spine.set_linewidth(1.1)


# ============================================================
# 4. 绘制底图和国家填色
# ============================================================

# 先画所有陆地为白色，保证南极为白色
world.plot(
    ax=ax,
    color="white",
    edgecolor="white",
    linewidth=0.25,
    zorder=1
)

# 再绘制非南极国家分级填色
world_no_antarctica = world_plot[world_plot["ADMIN"] != "Antarctica"].copy()

world_no_antarctica.plot(
    ax=ax,
    column="MeHg_export",
    cmap=blue_cmap,
    norm=norm,
    edgecolor="white",
    linewidth=0.35,
    zorder=2
)


# ============================================================
# 5. 绘制 MeHg yield 气泡
# ============================================================

ax.scatter(
    points["lon"],
    points["lat"],
    s=yield_to_size(points["MeHg_yield"]),
    facecolor="#ffa7a0",
    edgecolor="#ff6d6d",
    linewidth=0.6,
    alpha=0.68,
    zorder=5
)


# ============================================================
# 6. 左侧气泡大小图例
# ============================================================

legend_x = -162
legend_y = 12

ax.text(
    legend_x - 8,
    legend_y + 18,
    "MeHg yield\n(ng m$^{-2}$ yr$^{-1}$)",
    ha="left",
    va="center",
    fontsize=11,
    linespacing=1.0,
    zorder=10
)

legend_levels = [75, 225, 375, 525, 750]
legend_labels = ["0–150", "150–300", "300–450", "450–600", ">600"]

for i, (val, label) in enumerate(zip(legend_levels, legend_labels)):
    yy = legend_y - i * 12

    ax.scatter(
        [legend_x],
        [yy],
        s=yield_to_size(val),
        facecolor="white",
        edgecolor="black",
        linewidth=0.65,
        zorder=10
    )

    ax.text(
        legend_x + 10,
        yy,
        label,
        ha="left",
        va="center",
        fontsize=10.5,
        zorder=10
    )


# ============================================================
# 7. 底部横向 colorbar
# ============================================================

cax = fig.add_axes([0.18, 0.10, 0.68, 0.045])

sm = ScalarMappable(norm=norm, cmap=blue_cmap)
sm.set_array([])

cbar = fig.colorbar(
    sm,
    cax=cax,
    orientation="horizontal"
)

cbar.set_ticks([0.01, 0.1, 1, 10, 100, 1000])
cbar.ax.xaxis.set_major_formatter(
    FuncFormatter(lambda x, pos: "1,000" if x == 1000 else f"{x:g}")
)

cbar.ax.tick_params(
    labelsize=10.5,
    direction="out",
    length=6,
    width=0.8,
    pad=5
)

cbar.outline.set_linewidth(0.8)
cbar.ax.set_title(
    r"MeHg export (kg yr$^{-1}$)",
    fontsize=11,
    pad=10
)


# ============================================================
# 8. 保存与显示
# ============================================================

plt.savefig(
    "reproduced_mehg_world_map.png",
    dpi=300,
    bbox_inches="tight",
    pad_inches=0.04
)

plt.close(fig)
```

## 四、可调整参数说明

控制国家填色范围：

```python
norm = LogNorm(vmin=0.01, vmax=3000)
```

控制蓝色渐变：

```python
blue_cmap = LinearSegmentedColormap.from_list(
    "meh_export_blue",
    ["#74aef2", "#5d9fe6", "#3f86d1", "#216db8", "#0b55a5", "#003f88"]
)
```

控制气泡大小：

```python
def yield_to_size(y):
    return 8 + (np.asarray(y) / 600) ** 1.45 * 260
```

如果气泡偏大，可改为：

```python
return 6 + (np.asarray(y) / 600) ** 1.35 * 190
```

控制气泡颜色：

```python
facecolor="#ffa7a0"
edgecolor="#ff6d6d"
alpha=0.68
```
