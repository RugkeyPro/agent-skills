一、图像类型识别与结构拆解

这是一个**水平堆叠条形图 / Horizontal stacked bar chart**，展示 2021 年不同国家的风电基础设施数量。

主要特征：

- 横轴：数量规模，约 `0–4500`
    
- 纵轴：国家名称
    
- 深蓝色：`wind`
    
- 橙色：`probable wind`
    
- 数值标签：显示每个国家的总量
    
- 标题：`Wind Infrastructure 2021`
    
- 图例：右下方，横向色块图例
    
- 风格：白底、淡灰色纵向网格、大字号、无明显外框

## 二、单独的虚拟数据生成代码

```python
import pandas as pd


def generate_wind_infrastructure_data():
    """
    生成与原图接近的虚拟数据。
    
    total: 图中显示的总数
    wind: 已确认 wind 数量
    probable_wind: probable wind 数量
    """

    df = pd.DataFrame({
        "country": [
            "China",
            "United Kingdom",
            "Germany",
            "Denmark",
            "Netherlands",
            "Belgium"
        ],
        "total": [4359, 2591, 1519, 590, 469, 407],
        "probable_wind": [170, 18, 0, 0, 0, 0]
    })

    df["wind"] = df["total"] - df["probable_wind"]

    return df


df = generate_wind_infrastructure_data()
print(df)
```

## 三、完整 Python 绘图代码

```python
import pandas as pd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
matplotlib.rcParams["svg.fonttype"] = "none"
from matplotlib.patches import Patch


# ============================================================
# 1. 虚拟数据
# ============================================================

def generate_wind_infrastructure_data():
    df = pd.DataFrame({
        "country": [
            "China",
            "United Kingdom",
            "Germany",
            "Denmark",
            "Netherlands",
            "Belgium"
        ],
        "total": [4359, 2591, 1519, 590, 469, 407],
        "probable_wind": [170, 18, 0, 0, 0, 0]
    })

    df["wind"] = df["total"] - df["probable_wind"]
    return df


df = generate_wind_infrastructure_data()


# ============================================================
# 2. 基础样式
# ============================================================

plt.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Arial", "DejaVu Sans"],
    "axes.edgecolor": "none",
    "xtick.direction": "out",
    "ytick.direction": "out",
    "savefig.dpi": 300,
})

color_wind = "#00445d"
color_probable = "#ffa600"
text_color = "#00445d"
grid_color = "#d9d9d9"


# ============================================================
# 3. 绘图
# ============================================================

fig, ax = plt.subplots(figsize=(9.0, 5.2), dpi=120)

y = range(len(df))
bar_height = 0.66

# wind 主体条形
ax.barh(
    y,
    df["wind"],
    height=bar_height,
    color=color_wind,
    edgecolor=color_wind,
    label="wind",
    zorder=3
)

# probable wind 橙色堆叠段
ax.barh(
    y,
    df["probable_wind"],
    left=df["wind"],
    height=bar_height,
    color=color_probable,
    edgecolor=color_probable,
    label="probable wind",
    zorder=4
)


# ============================================================
# 4. 数值标签
# ============================================================

for i, total in enumerate(df["total"]):
    ax.text(
        total + 55,
        i,
        f"{total}",
        ha="left",
        va="center",
        fontsize=18,
        color=text_color
    )


# ============================================================
# 5. 坐标轴、网格和标题
# ============================================================

ax.set_yticks(list(y))
ax.set_yticklabels(df["country"], fontsize=20)
ax.invert_yaxis()

ax.set_xlim(0, 4650)
ax.set_xticks([0, 1000, 2000, 3000, 4000])
ax.tick_params(axis="x", labelsize=20, length=0, pad=12)
ax.tick_params(axis="y", length=0, pad=12)

ax.set_title(
    "Wind Infrastructure 2021",
    fontsize=22,
    pad=8
)

ax.grid(
    axis="x",
    color=grid_color,
    linewidth=1.4,
    zorder=0
)

ax.grid(axis="y", visible=False)

for spine in ax.spines.values():
    spine.set_visible(False)


# ============================================================
# 6. 图例
# ============================================================

legend_handles = [
    Patch(facecolor=color_wind, edgecolor=color_wind, label="wind"),
    Patch(facecolor=color_probable, edgecolor=color_probable, label="probable wind")
]

ax.legend(
    handles=legend_handles,
    loc="lower center",
    bbox_to_anchor=(0.67, 0.02),
    frameon=False,
    fontsize=20,
    handlelength=1.8,
    handletextpad=0.8,
    labelspacing=0.7
)


# ============================================================
# 7. 布局与保存
# ============================================================

plt.subplots_adjust(
    left=0.28,
    right=0.95,
    top=0.86,
    bottom=0.18
)

plt.savefig(
    "reproduced_wind_infrastructure_2021.png",
    dpi=300,
    bbox_inches="tight",
    pad_inches=0.04
)

plt.close(fig)
```

## 四、可调整参数说明

控制条形粗细：

```python
bar_height = 0.66
```

控制深蓝和橙色：

```python
color_wind = "#00445d"
color_probable = "#ffa600"
```

控制数值标签位置：

```python
total + 55
```

如果标签离条形太远，可改为：

```python
total + 30
```

控制图例位置：

```python
bbox_to_anchor=(0.67, 0.02)
```
