一、图像类型识别与结构拆解

这是一张**水平堆叠条形图 / Horizontal stacked bar chart**，用于展示不同 EEZ 国家或地区的非捕捞活动中，**公开追踪**与**未公开追踪**两类活动的组成。

图像结构为：

```text
横轴：Non-fishing activity 数量
纵轴：EEZ 国家 / 地区
红色段：not publicly tracked
蓝色段：publicly tracked
右侧红色百分比：not publicly tracked 占比
标题：Non-fishing Activity by EEZ
图例：底部右侧，红蓝色块
```

视觉特征：

- 国家按总活动量从高到低排列。
    
- 红色部分始终位于条形左端，蓝色部分堆叠在右侧。
    
- China 总量最高，蓝色公开追踪部分占主导。
    
- Indonesia、Vietnam、Thailand、Myanmar 等红色占比较高。
    
- 背景使用浅灰色竖向网格线。
    
- 坐标轴和边框弱化，整体为信息图风格。

## 二、单独的虚拟数据生成代码

```python
import pandas as pd


def generate_non_fishing_activity_data():
    """
    生成用于复现图像的虚拟数据。

    total_activity: 非捕捞活动总量
    not_public_pct: 未公开追踪活动占比，单位 %
    not_publicly_tracked: 未公开追踪数量
    publicly_tracked: 公开追踪数量
    """

    data = {
        "EEZ": [
            "China", "Indonesia", "Japan", "Malaysia", "United States",
            "Philippines", "India", "Vietnam", "Brazil", "South Korea",
            "United Kingdom", "Thailand", "Australia", "Mexico",
            "Emirates", "Turkey", "Spain", "Greece", "Myanmar", "Italy"
        ],
        "total_activity": [
            7400, 3950, 1980, 1040, 1010,
            900, 840, 710, 720, 700,
            660, 650, 560, 520,
            455, 450, 420, 400, 380, 370
        ],
        "not_public_pct": [
            15, 53, 11, 45, 19,
            46, 33, 80, 11, 14,
            8, 70, 16, 23,
            15, 11, 5, 7, 92, 7
        ]
    }

    df = pd.DataFrame(data)

    df["not_publicly_tracked"] = df["total_activity"] * df["not_public_pct"] / 100
    df["publicly_tracked"] = df["total_activity"] - df["not_publicly_tracked"]

    return df


df = generate_non_fishing_activity_data()
print(df.head())
```

## 三、完整 Python 绘图代码

```python
import pandas as pd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
matplotlib.rcParams["svg.fonttype"] = "none"
from matplotlib.patches import Patch


# ============================================================
# 1. 虚拟数据
# ============================================================

def generate_non_fishing_activity_data():
    data = {
        "EEZ": [
            "China", "Indonesia", "Japan", "Malaysia", "United States",
            "Philippines", "India", "Vietnam", "Brazil", "South Korea",
            "United Kingdom", "Thailand", "Australia", "Mexico",
            "Emirates", "Turkey", "Spain", "Greece", "Myanmar", "Italy"
        ],
        "total_activity": [
            7400, 3950, 1980, 1040, 1010,
            900, 840, 710, 720, 700,
            660, 650, 560, 520,
            455, 450, 420, 400, 380, 370
        ],
        "not_public_pct": [
            15, 53, 11, 45, 19,
            46, 33, 80, 11, 14,
            8, 70, 16, 23,
            15, 11, 5, 7, 92, 7
        ]
    }

    df = pd.DataFrame(data)
    df["not_publicly_tracked"] = df["total_activity"] * df["not_public_pct"] / 100
    df["publicly_tracked"] = df["total_activity"] - df["not_publicly_tracked"]

    return df


df = generate_non_fishing_activity_data()


# ============================================================
# 2. 基础样式
# ============================================================

plt.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Arial", "DejaVu Sans"],
    "axes.edgecolor": "none",
    "xtick.direction": "out",
    "ytick.direction": "out",
    "savefig.dpi": 300,
})

color_not_public = "#d6002a"
color_public = "#2c7fb8"
grid_color = "#d9d9d9"
text_color = "#2b2b2b"


# ============================================================
# 3. 绘图
# ============================================================

fig, ax = plt.subplots(figsize=(7.3, 10.4), dpi=120)

y = range(len(df))
bar_height = 0.66

# 未公开追踪：红色左侧段
ax.barh(
    y,
    df["not_publicly_tracked"],
    height=bar_height,
    color=color_not_public,
    edgecolor=color_not_public,
    zorder=3,
    label="not publicly tracked"
)

# 公开追踪：蓝色右侧段
ax.barh(
    y,
    df["publicly_tracked"],
    left=df["not_publicly_tracked"],
    height=bar_height,
    color=color_public,
    edgecolor=color_public,
    zorder=3,
    label="publicly tracked"
)


# ============================================================
# 4. 百分比标签
# ============================================================

for i, row in df.iterrows():
    ax.text(
        row["total_activity"] + 100,
        i,
        f"{int(row['not_public_pct'])}%",
        ha="left",
        va="center",
        fontsize=16,
        color=color_not_public
    )


# ============================================================
# 5. 坐标轴与标题
# ============================================================

ax.set_yticks(list(y))
ax.set_yticklabels(df["EEZ"], fontsize=18, color=text_color)
ax.invert_yaxis()

ax.set_xlim(0, 7900)
ax.set_xticks([0, 2000, 4000, 6000])
ax.tick_params(axis="x", labelsize=17, length=0, pad=8, colors=text_color)
ax.tick_params(axis="y", length=0, pad=10)

ax.set_title(
    "Non-fishing Activity by EEZ",
    fontsize=22,
    pad=8,
    color=text_color
)

ax.grid(
    axis="x",
    color=grid_color,
    linewidth=1.3,
    zorder=0
)

ax.grid(axis="y", visible=False)

for spine in ax.spines.values():
    spine.set_visible(False)


# ============================================================
# 6. 图例
# ============================================================

legend_handles = [
    Patch(facecolor=color_not_public, edgecolor=color_not_public, label="not publicly tracked"),
    Patch(facecolor=color_public, edgecolor=color_public, label="publicly tracked")
]

ax.legend(
    handles=legend_handles,
    loc="lower right",
    bbox_to_anchor=(0.98, 0.02),
    frameon=False,
    fontsize=17,
    handlelength=1.8,
    handletextpad=0.8,
    labelspacing=0.6
)


# ============================================================
# 7. 保存与显示
# ============================================================

plt.subplots_adjust(
    left=0.34,
    right=0.96,
    top=0.94,
    bottom=0.07
)

plt.savefig(
    "reproduced_non_fishing_activity_by_eez.png",
    dpi=300,
    bbox_inches="tight",
    pad_inches=0.04
)

plt.close(fig)
```
