一、图像类型识别与结构拆解

这是一张**双面板组合图**：

```text
Panel b：二维密度散点图 / density scatter plot
Panel c：两组柱状图 / two-category bar chart
```

图像结构：

- **Panel b**
    
    - 横轴：`Habitat hotspot value`
        
    - 纵轴：`Climate change magnitude`
        
    - 数据点非常密集，点颜色表示局部密度。
        
    - 颜色图例嵌入在图内右下方，标题为 `Density`，下方标注 `Low` 和 `High`。
        
    - 使用类似 `viridis` 的连续色带：低密度为深紫色，高密度为黄绿色。
        
- **Panel c**
    
    - 横轴：`Unprotected`、`Protected`
        
    - 纵轴：`Habitat proportion`
        
    - 两个柱分别为蓝色和红橙色。
        
    - 柱顶标注百分比：`40.1%` 和 `59.9%`。

## 二、单独的虚拟数据生成代码

```python
import numpy as np
import pandas as pd


def generate_habitat_hotspot_data(seed=42, n=65000):
    """
    生成 Panel b 的虚拟二维散点数据。

    habitat_hotspot_value:
        栖息地热点值，范围 0-1，大量样本集中在 0 附近。

    climate_change_magnitude:
        气候变化幅度，范围 0-1，整体形成左侧高密度、右侧收缩的楔形分布。
    """
    rng = np.random.default_rng(seed)

    # x 大量集中在低值区，少量延伸到 1
    x_main = rng.beta(0.45, 5.8, int(n * 0.88))
    x_tail = rng.beta(1.8, 4.2, n - len(x_main))
    x = np.concatenate([x_main, x_tail])
    rng.shuffle(x)

    # 构造随 x 增大而逐渐收窄的 y 分布
    center = 0.52 + 0.05 * np.sin(4.5 * x)
    spread = 0.28 * np.exp(-2.4 * x) + 0.055

    y = center + rng.normal(0, spread, n)

    # 增加左侧近零 x 的垂直高密度带
    left = x < 0.04
    y[left] = rng.beta(2.1, 2.1, left.sum())

    # 形成轻微上边界和下边界收缩
    y += 0.10 * np.exp(-((x - 0.12) / 0.10) ** 2) * rng.normal(0, 0.7, n)
    y = np.clip(y, 0, 1)

    df_scatter = pd.DataFrame({
        "habitat_hotspot_value": x,
        "climate_change_magnitude": y
    })

    return df_scatter


def generate_habitat_proportion_data():
    """
    生成 Panel c 的柱状图数据。
    """
    df_bar = pd.DataFrame({
        "category": ["Unprotected", "Protected"],
        "habitat_proportion": [0.401, 0.599]
    })

    return df_bar


if __name__ == "__main__":
    scatter_df = generate_habitat_hotspot_data()
    bar_df = generate_habitat_proportion_data()

    scatter_df.to_csv("panel_b_density_scatter.csv", index=False)
    bar_df.to_csv("panel_c_habitat_proportion.csv", index=False)

    print(scatter_df.head())
    print(bar_df)
```

## 三、完整 Python 绘图代码

```python
import numpy as np
import pandas as pd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
matplotlib.rcParams["svg.fonttype"] = "none"
import matplotlib as mpl
from matplotlib.colors import Normalize
from mpl_toolkits.axes_grid1.inset_locator import inset_axes


# ============================================================
# 1. 虚拟数据生成
# ============================================================

def generate_habitat_hotspot_data(seed=42, n=65000):
    rng = np.random.default_rng(seed)

    x_main = rng.beta(0.45, 5.8, int(n * 0.88))
    x_tail = rng.beta(1.8, 4.2, n - len(x_main))
    x = np.concatenate([x_main, x_tail])
    rng.shuffle(x)

    center = 0.52 + 0.05 * np.sin(4.5 * x)
    spread = 0.28 * np.exp(-2.4 * x) + 0.055

    y = center + rng.normal(0, spread, n)

    left = x < 0.04
    y[left] = rng.beta(2.1, 2.1, left.sum())

    y += 0.10 * np.exp(-((x - 0.12) / 0.10) ** 2) * rng.normal(0, 0.7, n)
    y = np.clip(y, 0, 1)

    return pd.DataFrame({
        "habitat_hotspot_value": x,
        "climate_change_magnitude": y
    })


def generate_habitat_proportion_data():
    return pd.DataFrame({
        "category": ["Unprotected", "Protected"],
        "habitat_proportion": [0.401, 0.599]
    })


scatter_df = generate_habitat_hotspot_data()
bar_df = generate_habitat_proportion_data()


# ============================================================
# 2. 计算散点局部密度
# ============================================================

def assign_point_density(x, y, xlim=(0, 1), ylim=(0, 1), bins=180):
    """
    使用二维直方图为每个点分配局部密度值。
    """
    counts, xedges, yedges = np.histogram2d(
        x,
        y,
        bins=bins,
        range=[xlim, ylim]
    )

    xi = np.searchsorted(xedges, x, side="right") - 1
    yi = np.searchsorted(yedges, y, side="right") - 1

    xi = np.clip(xi, 0, bins - 1)
    yi = np.clip(yi, 0, bins - 1)

    density = counts[xi, yi]
    density = density / density.max()

    return density


scatter_df["density"] = assign_point_density(
    scatter_df["habitat_hotspot_value"].to_numpy(),
    scatter_df["climate_change_magnitude"].to_numpy()
)

# 为了让高密度点显示在上层，按密度排序
scatter_df = scatter_df.sort_values("density")


# ============================================================
# 3. 基础样式与配色
# ============================================================

plt.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Arial", "DejaVu Sans"],
    "axes.linewidth": 1.0,
    "xtick.direction": "out",
    "ytick.direction": "out",
    "xtick.major.size": 4,
    "ytick.major.size": 4,
    "savefig.dpi": 300,
})

density_cmap = plt.get_cmap("viridis")

bar_colors = {
    "Unprotected": "#4169d8",
    "Protected": "#f23b00",
}

grid_color = "#e6e6e6"
axis_color = "#444444"
text_color = "#000000"


# ============================================================
# 4. 创建画布
# ============================================================

fig, axes = plt.subplots(
    nrows=1,
    ncols=2,
    figsize=(7.0, 3.35),
    dpi=150,
    gridspec_kw={
        "width_ratios": [1.35, 1.0],
        "wspace": 0.40
    }
)

ax_b, ax_c = axes


# ============================================================
# 5. Panel b：二维密度散点图
# ============================================================

sc = ax_b.scatter(
    scatter_df["habitat_hotspot_value"],
    scatter_df["climate_change_magnitude"],
    c=scatter_df["density"],
    cmap=density_cmap,
    s=1.0,
    linewidths=0,
    alpha=0.82,
    rasterized=True
)

ax_b.set_xlim(-0.02, 1.08)
ax_b.set_ylim(-0.05, 1.05)

ax_b.set_xlabel("Habitat hotspot value", fontsize=14, labelpad=9)
ax_b.set_ylabel("Climate change magnitude", fontsize=14)

ax_b.set_xticks([0, 0.25, 0.50, 0.75, 1.00])
ax_b.set_yticks([0, 0.25, 0.50, 0.75, 1.00])

ax_b.set_xticklabels(["0", "0.25", "0.50", "0.75", "1.00"])
ax_b.set_yticklabels(["0", "0.25", "0.50", "0.75", "1.00"])

ax_b.grid(True, color=grid_color, linewidth=0.6, alpha=0.65)

for spine in ax_b.spines.values():
    spine.set_color(axis_color)
    spine.set_linewidth(1.0)

ax_b.tick_params(labelsize=11)

ax_b.text(
    -0.24,
    1.06,
    "b",
    transform=ax_b.transAxes,
    fontsize=16,
    fontweight="bold",
    ha="left",
    va="bottom"
)

# 图内水平 colorbar
cax = inset_axes(
    ax_b,
    width="47%",
    height="10%",
    loc="lower right",
    bbox_to_anchor=(-0.03, 0.11, 1, 1),
    bbox_transform=ax_b.transAxes,
    borderpad=0.6
)

norm = Normalize(vmin=0, vmax=1)
sm = mpl.cm.ScalarMappable(norm=norm, cmap=density_cmap)
sm.set_array([])

cb = plt.colorbar(
    sm,
    cax=cax,
    orientation="horizontal"
)

cb.set_ticks([])
cb.outline.set_visible(False)

ax_b.text(
    0.74,
    0.33,
    "Density",
    transform=ax_b.transAxes,
    fontsize=13,
    ha="center",
    va="center"
)

ax_b.text(
    0.48,
    0.09,
    "Low",
    transform=ax_b.transAxes,
    fontsize=11,
    ha="center",
    va="center"
)

ax_b.text(
    0.92,
    0.09,
    "High",
    transform=ax_b.transAxes,
    fontsize=11,
    ha="center",
    va="center"
)


# ============================================================
# 6. Panel c：柱状图
# ============================================================

x = np.arange(len(bar_df))
values = bar_df["habitat_proportion"].to_numpy()
labels = bar_df["category"].to_list()

bars = ax_c.bar(
    x,
    values,
    width=0.88,
    color=[bar_colors[label] for label in labels],
    edgecolor="#333333",
    linewidth=1.0,
    zorder=3
)

for i, bar in enumerate(bars):
    v = values[i]
    ax_c.text(
        bar.get_x() + bar.get_width() / 2,
        v + 0.015,
        f"{v * 100:.1f}%",
        ha="center",
        va="bottom",
        fontsize=13,
        color=text_color
    )

ax_c.set_ylim(0, 1.05)
ax_c.set_xlim(-0.6, 1.6)

ax_c.set_ylabel("Habitat proportion", fontsize=14)
ax_c.set_xticks(x)
ax_c.set_xticklabels(labels, fontsize=11)

ax_c.set_yticks([0, 0.25, 0.50, 0.75, 1.00])
ax_c.set_yticklabels(["0", "0.25", "0.50", "0.75", "1.00"])

ax_c.grid(True, color=grid_color, linewidth=0.6, alpha=0.65, zorder=0)

for spine in ax_c.spines.values():
    spine.set_color(axis_color)
    spine.set_linewidth(1.0)

ax_c.tick_params(labelsize=11)

ax_c.text(
    -0.38,
    1.06,
    "c",
    transform=ax_c.transAxes,
    fontsize=16,
    fontweight="bold",
    ha="left",
    va="bottom"
)


# ============================================================
# 7. 保存与显示
# ============================================================

plt.subplots_adjust(
    left=0.095,
    right=0.98,
    bottom=0.25,
    top=0.92,
    wspace=0.42
)

plt.savefig(
    "reproduced_habitat_hotspot_density_bar.png",
    dpi=300,
    bbox_inches="tight",
    pad_inches=0.04
)

plt.close(fig)
```
