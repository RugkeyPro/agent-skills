```text

c：预测值-观测值散点图，带 1:1 参考线和训练/测试统计指标
d：双 y 轴直方图 + 累积分布曲线

f：RMU 与化学结构差异的批次级关系散点图
```

## 二、单独的虚拟数据生成代码

```python
import numpy as np
import pandas as pd


def generate_panel_a_data(seed=42):
    rng = np.random.default_rng(seed)

    n_std = 3556
    n_non = 514

    reported_std = np.concatenate([
        rng.normal(1.2, 0.75, int(n_std * 0.94)),
        rng.normal(-3.0, 1.2, n_std - int(n_std * 0.94))
    ])
    predicted_std = rng.normal(1.0, 0.55, n_std)

    reported_non = np.concatenate([
        rng.normal(1.25, 0.80, int(n_non * 0.96)),
        rng.normal(-1.2, 0.8, n_non - int(n_non * 0.96))
    ])
    predicted_non = rng.normal(1.05, 0.50, n_non)

    data = []

    for v in reported_std:
        data.append(["standardized", "reported POD", v])
    for v in predicted_std:
        data.append(["standardized", "predicted POD", v])
    for v in reported_non:
        data.append(["non-standardized", "reported POD", v])
    for v in predicted_non:
        data.append(["non-standardized", "predicted POD", v])

    df = pd.DataFrame(data, columns=["subset", "type", "POD"])
    df["POD"] = df["POD"].clip(-6.8, 5.8)

    return df


def generate_panel_b_data():
    return pd.DataFrame({
        "subset": ["standardized", "standardized", "non-standardized", "non-standardized"],
        "metric": ["observed prediction error (RMSE)", "prediction uncertainty (RMU)",
                   "observed prediction error (RMSE)", "prediction uncertainty (RMU)"],
        "value": [0.72, 0.85, 1.06, 1.00]
    })


def generate_panel_c_data(seed=7):
    rng = np.random.default_rng(seed)

    n_train = 257
    n_test = 257

    pred_train = rng.normal(0.4, 0.85, n_train)
    reported_train = 0.70 * pred_train + rng.normal(0, 0.55, n_train) + 0.3

    pred_test = rng.normal(0.7, 0.85, n_test)
    reported_test = 0.75 * pred_test + rng.normal(0, 0.75, n_test) + 0.5

    df_train = pd.DataFrame({
        "subset": "Training data",
        "predicted_POD": pred_train,
        "reported_POD": reported_train
    })

    df_test = pd.DataFrame({
        "subset": "Test data (CV)",
        "predicted_POD": pred_test,
        "reported_POD": reported_test
    })

    return pd.concat([df_train, df_test], ignore_index=True)


def generate_panel_d_data(seed=12):
    rng = np.random.default_rng(seed)

    std = rng.gamma(shape=7.0, scale=0.28, size=1778) + 1.10
    non = rng.gamma(shape=7.5, scale=0.32, size=257) + 1.30

    std = np.clip(std, 1.0, 8.0)
    non = np.clip(non, 1.0, 8.2)

    return pd.DataFrame({
        "subset": ["standardized"] * len(std) + ["non-standardized"] * len(non),
        "uncertainty": np.concatenate([std, non])
    })


def generate_panel_e_data(seed=21):
    rng = np.random.default_rng(seed)
    n = 16

    rmu = np.clip(rng.normal(0.95, 0.22, n), 0.45, 1.75)
    rmse = 0.95 * rmu + rng.normal(0, 0.18, n)
    rmse = np.clip(rmse, 0.25, 1.75)

    # 加几个高值点，使形态接近原图
    rmu[0], rmse[0] = 1.56, 1.55
    rmu[1], rmse[1] = 0.95, 1.45

    return pd.DataFrame({
        "RMU": rmu,
        "RMSE": rmse
    })


def generate_panel_f_data(seed=33):
    rng = np.random.default_rng(seed)
    n = 16

    rmu = np.clip(rng.normal(0.95, 0.22, n), 0.45, 1.75)
    dissimilarity = 0.15 + 0.50 * (1 - np.exp(-2.7 * (rmu - 0.45))) + rng.normal(0, 0.06, n)
    dissimilarity = np.clip(dissimilarity, 0.0, 0.9)

    rmu[0], dissimilarity[0] = 1.55, 0.72
    rmu[1], dissimilarity[1] = 0.58, 0.32

    return pd.DataFrame({
        "RMU": rmu,
        "jaccard_dissimilarity": dissimilarity
    })


if __name__ == "__main__":
    panel_a = generate_panel_a_data()
    panel_b = generate_panel_b_data()
    panel_c = generate_panel_c_data()
    panel_d = generate_panel_d_data()
    panel_e = generate_panel_e_data()
    panel_f = generate_panel_f_data()

    panel_a.to_csv("panel_a_violin.csv", index=False)
    panel_b.to_csv("panel_b_bar.csv", index=False)
    panel_c.to_csv("panel_c_prediction.csv", index=False)
    panel_d.to_csv("panel_d_uncertainty.csv", index=False)
    panel_e.to_csv("panel_e_rmu_rmse.csv", index=False)
    panel_f.to_csv("panel_f_rmu_dissimilarity.csv", index=False)
```
