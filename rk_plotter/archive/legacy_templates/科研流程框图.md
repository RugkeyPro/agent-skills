一、图像类型识别与结构拆解

这是一个**科研流程框图 / 模型耦合机制图 / 风险评估流程图**，不是统计图。  
核心结构是：

- 外层蓝色矩形边界：表示整体建模系统范围。
    
- 顶部蓝色输入线：表示“大气沉降与点源/非点源污染输入”。
    
- 左侧模块链：水动力过程链  
    `Rainfall runoff model → Hydraulic model → 3D-Hydro model`
    
- 右侧模块链：污染物负荷过程链  
    `Rainfall runoff model → Hydraulic model → 3D-Hydro model`
    
- 底部绿色椭圆：`3D-PFAS model`
    
- 右侧红色椭圆：`Risk assessment`
    
- 主要图形元素：圆角矩形、椭圆、箭头、边界框、流程文字标注。

# 2. 绘图函数

def draw_round_box(ax, xy, width, height, text, fill, edge):
    x, y = xy
    patch = FancyBboxPatch(
        (x - width / 2, y - height / 2),
        width,
        height,
        boxstyle="round,pad=0.02,rounding_size=1.2",
        linewidth=1.25,
        edgecolor=edge,
        facecolor=fill
    )
    ax.add_patch(patch)
    ax.text(x, y, text, ha="center", va="center", fontsize=15.5, linespacing=0.9)


def draw_ellipse(ax, xy, width, height, text, fill, edge, lw=1.25):
    x, y = xy
    patch = Ellipse(
        xy,
        width,
        height,
        linewidth=lw,
        edgecolor=edge,
        facecolor=fill
    )
    ax.add_patch(patch)
    ax.text(x, y, text, ha="center", va="center", fontsize=15.5, linespacing=0.9)


def draw_arrow(ax, start, end, color="black", lw=1.25, mutation_scale=12):
    arrow = FancyArrowPatch(
        start,
        end,
        arrowstyle="-|>",
        mutation_scale=mutation_scale,
        linewidth=lw,
        color=color,
        shrinkA=0,
        shrinkB=0
    )
    ax.add_patch(arrow)


def draw_flowchart(output_path="reproduced_flowchart.png"):
    nodes, arrows, labels = generate_flowchart_data()

    plt.rcParams.update({
        "font.family": "sans-serif",
        "font.sans-serif": ["Arial", "DejaVu Sans"],
        "figure.dpi": 120,
        "savefig.dpi": 300
    })

    fig, ax = plt.subplots(figsize=(8.94, 6.61))
    ax.set_xlim(0, 100)
    ax.set_ylim(0, 72)
    ax.set_aspect("equal")
    ax.axis("off")

    # 外层蓝色系统边界
    outer = Rectangle(
        (0.5, 0.5),
        74.5,
        70,
        linewidth=1.25,
        edgecolor="#0072 bc",
        facecolor="none"
    )
    ax.add_patch(outer)

    # 绘制普通箭头
    for item in arrows:
        color = item.get("color", "black")
        lw = 1.25 if color == "black" else 1.15
        draw_arrow(ax, item["start"], item["end"], color=color, lw=lw)

    # 底部 Direct model output 的折线路径
    ax.plot([27, 27], [18.4, 15.5], color="black", lw=1.25)
    ax.plot([27, 40.5], [15.5, 15.5], color="black", lw=1.25)
    draw_arrow(ax, (40.5, 15.5), (40.5, 10.2), color="black", lw=1.25)

    ax.plot([65, 65], [18.4, 15.5], color="black", lw=1.25)
    ax.plot([65, 51.5], [15.5, 15.5], color="black", lw=1.25)
    draw_arrow(ax, (51.5, 15.5), (51.5, 10.2), color="black", lw=1.25)

    # 右侧白色粗箭头，指向风险评估
    risk_arrow = FancyArrow(
        75.8, 36,
        4.6, 0,
        width=2.8,
        head_width=5.0,
        head_length=2.5,
        length_includes_head=True,
        facecolor="white",
        edgecolor="black",
        linewidth=1.25
    )
    ax.add_patch(risk_arrow)

    # 绘制节点
    for node in nodes:
        if node["type"] == "round_box":
            draw_round_box(
                ax,
                node["xy"],
                node["width"],
                node["height"],
                node["text"],
                node["fill"],
                node["edge"]
            )
        elif node["type"] == "ellipse":
            draw_ellipse(
                ax,
                node["xy"],
                node["width"],
                node["height"],
                node["text"],
                node["fill"],
                node["edge"],
                lw=1.35 if node["id"] == "risk" else 1.25
            )

    # 绘制文字标签
    for label in labels:
        ax.text(
            label["xy"][0],
            label["xy"][1],
            label["text"],
            ha=label.get("ha", "center"),
            va="center",
            fontsize=15.5,
            linespacing=0.95
        )

    plt.savefig(output_path, bbox_inches="tight", pad_inches=0.08)
    plt.close(fig)


# =========================
# 3. 执行绘图
# =========================

draw_flowchart("reproduced_flowchart.png")
