一、图像类型识别与结构拆解

这是一张**联合分布图 / Joint KDE plot / 预测值-真实值一致性图**。

图像由三部分组成：

```text
中心主图：truth vs imputed 的二维核密度等值线图
顶部边缘图：truth 的一维 KDE 分布
右侧边缘图：imputed 的一维 KDE 分布
```

主要元素：

- 横轴：`truth`
    
- 纵轴：`imputed`
    
- 两组方法：
    
    - 红色：`STIMP`
        
    - 蓝色：`DINEOF`
        
- 黑色斜线：`y = x` 一致性参考线
    
- 主图：二维 KDE 等值线，红色更贴近对角线，蓝色更分散
    
- 顶部和右侧：红色填充的边缘密度分布
    
- 图内标注：两种方法的 PCC 和 p-value
    
- 图例：左上角，标题为 `method`

## 二、单独的虚拟数据生成代码

```python
import numpy as np
import pandas as pd


def generate_joint_kde_validation_data(seed=42, n=420):
    """
    生成 truth-imputed 对比图的虚拟数据。

    字段：
    truth   : 真实值
    imputed : 填补值 / 预测值
    method  : 方法名称，STIMP 或 DINEOF
    """

    rng = np.random.default_rng(seed)

    # 构造偏右的 truth 分布，使其接近原图顶部边缘密度
    truth_core = rng.normal(0.45, 0.48, int(n * 0.72))
    truth_high = rng.normal(1.10, 0.28, int(n * 0.20))
    truth_low = rng.normal(-0.55, 0.22, n - len(truth_core) - len(truth_high))

    truth = np.concatenate([truth_core, truth_high, truth_low])
    rng.shuffle(truth)
    truth = np.clip(truth, -1.25, 2.05)

    # STIMP：高度贴近 y=x
    imputed_stimp = truth + rng.normal(0, 0.10, n)
    imputed_stimp += 0.03 * np.sin(2.2 * truth)

    # DINEOF：相关性较低，分布更宽，并带有轻微非线性偏差
    imputed_dineof = 0.78 * truth + 0.15 + rng.normal(0, 0.38, n)
    imputed_dineof += 0.12 * np.sin(2.8 * truth)

    df = pd.DataFrame({
        "truth": np.concatenate([truth, truth]),
        "imputed": np.concatenate([imputed_stimp, imputed_dineof]),
        "method": ["STIMP"] * n + ["DINEOF"] * n
    })

    return df


df = generate_joint_kde_validation_data()
print(df.head())
```

## 三、完整 Python 绘图代码

```python
import numpy as np
import pandas as pd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
matplotlib.rcParams["svg.fonttype"] = "none"

from scipy.stats import gaussian_kde, pearsonr
from matplotlib.gridspec import GridSpec
from matplotlib.lines import Line2D


# ============================================================
# 1. 虚拟数据生成
# ============================================================

def generate_joint_kde_validation_data(seed=42, n=420):
    rng = np.random.default_rng(seed)

    truth_core = rng.normal(0.45, 0.48, int(n * 0.72))
    truth_high = rng.normal(1.10, 0.28, int(n * 0.20))
    truth_low = rng.normal(-0.55, 0.22, n - len(truth_core) - len(truth_high))

    truth = np.concatenate([truth_core, truth_high, truth_low])
    rng.shuffle(truth)
    truth = np.clip(truth, -1.25, 2.05)

    imputed_stimp = truth + rng.normal(0, 0.10, n)
    imputed_stimp += 0.03 * np.sin(2.2 * truth)

    imputed_dineof = 0.78 * truth + 0.15 + rng.normal(0, 0.38, n)
    imputed_dineof += 0.12 * np.sin(2.8 * truth)

    return pd.DataFrame({
        "truth": np.concatenate([truth, truth]),
        "imputed": np.concatenate([imputed_stimp, imputed_dineof]),
        "method": ["STIMP"] * n + ["DINEOF"] * n
    })


df = generate_joint_kde_validation_data()


# ============================================================
# 2. KDE 工具函数
# ============================================================

def kde_1d(values, grid, bandwidth=0.22):
    kde = gaussian_kde(values, bw_method=bandwidth)
    return kde(grid)


def kde_2d(x, y, x_grid, y_grid, bandwidth=0.25):
    xx, yy = np.meshgrid(x_grid, y_grid)
    values = np.vstack([x, y])
    kde = gaussian_kde(values, bw_method=bandwidth)
    zz = kde(np.vstack([xx.ravel(), yy.ravel()])).reshape(xx.shape)
    return xx, yy, zz


def contour_levels_from_density(z, n_levels=8):
    """
    生成适合 KDE 等值线的 levels，避免极低密度区域画得过多。
    """
    zmax = z.max()
    return np.linspace(zmax * 0.12, zmax * 0.92, n_levels)


# ============================================================
# 3. 基础样式
# ============================================================

plt.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Arial", "DejaVu Sans"],
    "axes.linewidth": 1.0,
    "xtick.direction": "out",
    "ytick.direction": "out",
    "xtick.major.size": 3.5,
    "ytick.major.size": 3.5,
    "savefig.dpi": 300,
})

colors = {
    "STIMP": "#d7191c",
    "DINEOF": "#4575b4",
}

fill_red = "#c95b65"


# ============================================================
# 4. 画布布局：Joint plot 结构
# ============================================================

fig = plt.figure(figsize=(4.0, 4.2), dpi=140)

gs = GridSpec(
    nrows=2,
    ncols=2,
    width_ratios=[4.0, 0.85],
    height_ratios=[0.85, 4.0],
    wspace=0.04,
    hspace=0.04
)

ax_top = fig.add_subplot(gs[0, 0])
ax_main = fig.add_subplot(gs[1, 0])
ax_right = fig.add_subplot(gs[1, 1], sharey=ax_main)
ax_empty = fig.add_subplot(gs[0, 1])
ax_empty.axis("off")


# ============================================================
# 5. 坐标范围
# ============================================================

xlim = (-1.35, 2.20)
ylim = (-2.00, 2.65)

x_grid = np.linspace(*xlim, 260)
y_grid = np.linspace(*ylim, 260)


# ============================================================
# 6. 中心二维 KDE 等值线
# ============================================================

for method in ["DINEOF", "STIMP"]:
    sub = df[df["method"] == method]

    xx, yy, zz = kde_2d(
        sub["truth"].to_numpy(),
        sub["imputed"].to_numpy(),
        x_grid,
        y_grid,
        bandwidth=0.22 if method == "STIMP" else 0.28
    )

    ax_main.contour(
        xx,
        yy,
        zz,
        levels=contour_levels_from_density(zz, n_levels=8),
        colors=colors[method],
        linewidths=1.0,
        alpha=0.95
    )


# y = x 参考线
ax_main.plot(
    [-1.4, 2.2],
    [-1.4, 2.2],
    color="#555555",
    lw=1.0,
    zorder=1
)


# ============================================================
# 7. 顶部和右侧边缘 KDE
# ============================================================

stimp = df[df["method"] == "STIMP"]

top_density = kde_1d(stimp["truth"].to_numpy(), x_grid, bandwidth=0.23)
top_density = top_density / top_density.max()

ax_top.fill_between(
    x_grid,
    0,
    top_density,
    facecolor=fill_red,
    edgecolor=colors["STIMP"],
    linewidth=1.0,
    alpha=0.58
)

ax_top.plot(
    x_grid,
    top_density,
    color=colors["STIMP"],
    lw=0.9
)

right_density = kde_1d(stimp["imputed"].to_numpy(), y_grid, bandwidth=0.23)
right_density = right_density / right_density.max()

ax_right.fill_betweenx(
    y_grid,
    0,
    right_density,
    facecolor=fill_red,
    edgecolor=colors["STIMP"],
    linewidth=1.0,
    alpha=0.58
)

ax_right.plot(
    right_density,
    y_grid,
    color=colors["STIMP"],
    lw=0.9
)


# ============================================================
# 8. 相关系数文本
# ============================================================

r_stimp, p_stimp = pearsonr(
    df[df["method"] == "STIMP"]["truth"],
    df[df["method"] == "STIMP"]["imputed"]
)

r_dineof, p_dineof = pearsonr(
    df[df["method"] == "DINEOF"]["truth"],
    df[df["method"] == "DINEOF"]["imputed"]
)

ax_main.text(
    0.28,
    2.25,
    f"PCC={r_stimp:.4f} pvalue={p_stimp:.2f}",
    color=colors["STIMP"],
    fontsize=8,
    fontweight="bold"
)

ax_main.text(
    0.28,
    2.05,
    f"PCC={r_dineof:.4f} pvalue={p_dineof:.2f}",
    color=colors["DINEOF"],
    fontsize=8,
    fontweight="bold"
)


# ============================================================
# 9. 图例
# ============================================================

legend_handles = [
    Line2D([0], [0], color=colors["STIMP"], lw=1.0, label="STIMP"),
    Line2D([0], [0], color=colors["DINEOF"], lw=1.0, label="DINEOF"),
]

ax_main.legend(
    handles=legend_handles,
    title="method",
    loc="upper left",
    frameon=False,
    fontsize=6,
    title_fontsize=6,
    handlelength=2.0,
    borderpad=0.2,
    labelspacing=0.2
)


# ============================================================
# 10. 坐标轴、刻度和边框
# ============================================================

ax_main.set_xlim(xlim)
ax_main.set_ylim(ylim)

ax_main.set_xlabel("truth", fontsize=15)
ax_main.set_ylabel("imputed", fontsize=15)

ax_main.set_xticks([-1.0, -0.5, 0.0, 0.5, 1.0, 1.5, 2.0])
ax_main.set_yticks([-2, -1, 0, 1, 2])
ax_main.tick_params(labelsize=9)

# 顶部边缘图
ax_top.set_xlim(xlim)
ax_top.set_ylim(0, 1.15)
ax_top.set_xticklabels([])
ax_top.set_yticks([])
ax_top.tick_params(axis="x", length=2)

# 右侧边缘图
ax_right.set_xlim(0, 1.15)
ax_right.set_xticks([])
ax_right.set_yticklabels([])
ax_right.tick_params(axis="y", length=2)

# 去除多余边框，使其接近原图 jointplot 风格
ax_top.spines["right"].set_visible(False)
ax_top.spines["left"].set_visible(False)
ax_top.spines["top"].set_visible(False)

ax_right.spines["top"].set_visible(False)
ax_right.spines["bottom"].set_visible(False)
ax_right.spines["right"].set_visible(False)

for ax in [ax_main, ax_top, ax_right]:
    for spine in ax.spines.values():
        spine.set_color("#777777")
        spine.set_linewidth(0.8)

ax_main.spines["top"].set_visible(False)
ax_main.spines["right"].set_visible(False)


# ============================================================
# 11. 保存与显示
# ============================================================

plt.savefig(
    "reproduced_joint_kde_truth_imputed.png",
    dpi=300,
    bbox_inches="tight",
    pad_inches=0.04
)

plt.close(fig)
```

## 四、可调整参数说明

控制 STIMP 与 `y=x` 的贴近程度：

```python
imputed_stimp = truth + rng.normal(0, 0.10, n)
```

数值越小，红色等值线越窄，PCC 越接近 1。

控制 DINEOF 的离散程度：

```python
imputed_dineof = 0.78 * truth + 0.15 + rng.normal(0, 0.38, n)
```

增大 `0.38` 会让蓝色等值线更分散，PCC 降低。

控制二维 KDE 平滑程度：

```python
bandwidth=0.22 if method == "STIMP" else 0.28
```

数值越大，等值线越平滑；数值越小，等值线越贴合局部数据形态。
