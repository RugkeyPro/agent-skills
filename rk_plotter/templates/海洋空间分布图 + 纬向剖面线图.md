一、图像类型识别与结构拆解

这是一张**双面板海洋空间分布图 + 纬向剖面线图**。

左侧 **d**：

- 图类型：纬度剖面折线图
    
- y 轴：纬度，范围约 `60°S–60°N`
    
- x 轴：无具体数值标注，主要用于比较三条变量的纬向分布
    
- 三条线：
    
    - 蓝色：`Ingestion risk`
        
    - 橙色：`Organism biomass`
        
    - 紫色：`Ingestible plastic`
        
- 图例位于左图下方，横向偏左排列
    

右侧 **h**：

- 图类型：全球海洋栅格热图 / 世界地图空间分布图
    
- 变量：`Ingestion risk (mol C m⁻³ kg m⁻³)`
    
- 颜色：对数尺度，从 `10^-14` 到 `10^-10`
    
- 陆地：白色填充，黑色海岸线
    
- 海洋：彩色栅格，低值为紫蓝色，高值为红紫色
    
- 底部：水平 colorbar，左端带三角延伸

## 二、虚拟数据生成代码

```python
import numpy as np
import pandas as pd


def gaussian2d(lon, lat, lon0, lat0, sx, sy, amp):
    """二维高斯热点，用于模拟海洋风险高值区。"""
    dlon = np.minimum(np.abs(lon - lon0), 360 - np.abs(lon - lon0))
    return amp * np.exp(-(dlon ** 2 / (2 * sx ** 2) + (lat - lat0) ** 2 / (2 * sy ** 2)))


def generate_ingestion_risk_grid(seed=42):
    """
    生成全球海洋摄食风险的虚拟栅格数据。
    返回：
    lon2d, lat2d, risk
    risk 单位：mol C m^-3 kg m^-3
    """

    rng = np.random.default_rng(seed)

    lon = np.linspace(-180, 180, 721)
    lat = np.linspace(-70, 80, 301)
    lon2d, lat2d = np.meshgrid(lon, lat)

    # 背景：低纬和副热带略高，南大洋显著偏低
    log_risk = -12.7
    log_risk += 0.55 * np.exp(-(lat2d / 22) ** 2)
    log_risk += 0.45 * np.exp(-((lat2d - 25) / 18) ** 2)
    log_risk += 0.35 * np.exp(-((lat2d + 20) / 18) ** 2)
    log_risk -= 1.25 * np.exp(-((lat2d + 58) / 9) ** 2)

    # 经向带状差异
    log_risk += 0.25 * np.sin(np.deg2rad(lon2d * 1.8))
    log_risk += 0.18 * np.cos(np.deg2rad(lon2d * 3.4 + lat2d))

    # 主要热点，模拟北大西洋、北太平洋、印度洋、东南亚海域等
    hotspots = [
        (-45, 35, 22, 10, 1.25),
        (-150, 30, 28, 11, 1.05),
        (145, 30, 23, 10, 1.10),
        (80, 12, 30, 13, 0.95),
        (115, -10, 25, 11, 0.90),
        (15, 0, 24, 12, 0.65),
        (-80, -20, 20, 12, 0.45),
        (35, -25, 20, 10, 0.60),
    ]

    for lon0, lat0, sx, sy, amp in hotspots:
        log_risk += gaussian2d(lon2d, lat2d, lon0, lat0, sx, sy, amp)

    # 一些低值带，增强原图中条带状分布
    log_risk -= 0.70 * np.exp(-((lat2d + 42) / 8) ** 2) * (
        0.5 + 0.5 * np.sin(np.deg2rad(lon2d * 2.2))
    )
    log_risk -= 0.45 * np.exp(-((lat2d - 5) / 7) ** 2) * (
        0.5 + 0.5 * np.cos(np.deg2rad(lon2d * 2.7))
    )

    # 平滑随机纹理
    noise = rng.normal(0, 0.08, size=log_risk.shape)
    for _ in range(6):
        noise = (
            noise +
            np.roll(noise, 1, axis=0) +
            np.roll(noise, -1, axis=0) +
            np.roll(noise, 1, axis=1) +
            np.roll(noise, -1, axis=1)
        ) / 5

    log_risk += noise

    risk = 10 ** log_risk
    risk = np.clip(risk, 1e-14, 3e-10)

    return lon2d, lat2d, risk


def generate_latitudinal_profiles(lat):
    """
    生成左侧 panel d 的纬向剖面虚拟数据。
    三个变量均归一化到 0-1，用于复现形状。
    """

    ingestion = (
        0.30
        + 0.45 * np.exp(-((lat - 35) / 6) ** 2)
        + 0.28 * np.exp(-((lat - 15) / 13) ** 2)
        + 0.18 * np.exp(-((lat + 25) / 12) ** 2)
        + 0.06 * np.sin(np.deg2rad(lat * 8))
    )

    biomass = (
        0.16
        + 0.78 * np.exp(-((lat - 70) / 7) ** 2)
        + 0.20 * np.exp(-((lat - 10) / 20) ** 2)
        + 0.05 * np.cos(np.deg2rad(lat * 5))
    )

    plastic = (
        0.14
        + 0.82 * np.exp(-((lat - 30) / 13) ** 2)
        + 0.70 * np.exp(-((lat + 30) / 10) ** 2)
        + 0.25 * np.exp(-((lat - 5) / 15) ** 2)
    )

    def norm(x):
        return (x - x.min()) / (x.max() - x.min())

    df = pd.DataFrame({
        "latitude": lat,
        "Ingestion risk": norm(ingestion),
        "Organism biomass": norm(biomass),
        "Ingestible plastic": norm(plastic),
    })

    return df


if __name__ == "__main__":
    lon2d, lat2d, risk = generate_ingestion_risk_grid()
    profile_df = generate_latitudinal_profiles(np.linspace(-70, 80, 301))

    np.savez(
        "virtual_ingestion_risk_grid.npz",
        lon=lon2d,
        lat=lat2d,
        risk=risk
    )

    profile_df.to_csv("virtual_latitudinal_profiles.csv", index=False)
```
