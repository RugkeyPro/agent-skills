一、图像类型识别与结构拆解

这是一张**双面板时间序列折线图**，上下排列两个子图，分别标记为 **a** 和 **b**。  
核心元素包括：

- 横轴：日期，范围约为 `2013-03-01` 至 `2014-04-01`
    
- 纵轴：浓度，单位为 `ng L⁻¹`
    
- 每个子图包含 4 条 PFOA 情景折线和 1 条灰色 PNEC 水平线
    
- 上图 y 轴范围：`0–160`
    
- 下图 y 轴范围：`0–140`
    
- 配色：红、绿、橙、蓝、浅灰
    
- 图例位于整张图底部，带黑色边框

## 二、虚拟数据生成代码

```python
import numpy as np
import pandas as pd


def ar1_noise(n, rng, sigma=0.12, phi=0.82):
    """生成具有时间相关性的扰动，使折线具有原图那种连续但不完全平滑的波动。"""
    eps = rng.normal(0, sigma, n)
    out = np.zeros(n)
    for i in range(1, n):
        out[i] = phi * out[i - 1] + eps[i]
    return out


def add_gaussian_event(values, dates, center, amp, width_days):
    """添加局部峰值或谷值，用于模拟原图中尖峰和骤降。"""
    center = pd.Timestamp(center)
    days = (dates - center).days.values
    return values + amp * np.exp(-(days ** 2) / (2 * width_days ** 2))


def interpolate_base_series(dates, anchors):
    """根据关键日期锚点插值得到基础 PFOA_total 序列。"""
    x = dates.map(pd.Timestamp.toordinal).values
    anchor_dates = pd.to_datetime(list(anchors.keys()))
    anchor_x = anchor_dates.map(pd.Timestamp.toordinal).values
    anchor_y = np.array(list(anchors.values()), dtype=float)
    return np.interp(x, anchor_x, anchor_y)


def make_panel_data(panel_name, anchors, events, seed):
    rng = np.random.default_rng(seed)
    dates = pd.date_range("2013-03-01", "2014-04-05", freq="D")

    base = interpolate_base_series(dates, anchors)

    for center, amp, width in events:
        base = add_gaussian_event(base, dates, center, amp, width)

    base = base + ar1_noise(len(dates), rng, sigma=0.10, phi=0.86)
    base = np.clip(base, 4.6, None)

    df = pd.DataFrame({
        "panel": panel_name,
        "date": dates,
        "PFOA_total": base,
        "PFOA_total_2*loading": base * 2,
        "PFOA_total_5*loading": base * 5,
        "PFOA_total_10*loading": base * 10,
        "PNEC": 50
    })

    return df


anchors_a = {
    "2013-03-01": 8.2,
    "2013-04-01": 10.8,
    "2013-04-12": 12.3,
    "2013-05-01": 10.9,
    "2013-05-15": 13.4,
    "2013-06-01": 12.5,
    "2013-07-01": 8.6,
    "2013-08-01": 9.6,
    "2013-08-20": 11.6,
    "2013-09-10": 10.4,
    "2013-10-01": 8.2,
    "2013-11-01": 8.9,
    "2013-11-20": 10.2,
    "2013-12-05": 8.1,
    "2014-01-01": 9.5,
    "2014-02-01": 6.8,
    "2014-02-15": 5.0,
    "2014-03-01": 8.2,
    "2014-04-05": 10.6,
}

events_a = [
    ("2013-04-15", 1.0, 5),
    ("2013-05-06", -1.5, 4),
    ("2013-12-05", -1.3, 3),
    ("2014-02-15", -1.1, 4),
]


anchors_b = {
    "2013-03-01": 8.6,
    "2013-04-01": 10.2,
    "2013-04-12": 12.0,
    "2013-05-01": 10.1,
    "2013-05-12": 12.1,
    "2013-06-01": 11.8,
    "2013-07-01": 9.4,
    "2013-08-01": 9.8,
    "2013-08-20": 11.5,
    "2013-09-10": 10.2,
    "2013-10-01": 7.0,
    "2013-11-01": 8.7,
    "2013-11-20": 9.6,
    "2013-12-05": 8.0,
    "2014-01-10": 10.2,
    "2014-02-01": 6.9,
    "2014-02-15": 5.1,
    "2014-03-01": 8.4,
    "2014-04-05": 10.5,
}

events_b = [
    ("2013-04-12", 1.5, 5),
    ("2013-05-01", -1.5, 4),
    ("2013-08-20", 0.9, 6),
    ("2013-09-25", -1.2, 6),
    ("2013-12-05", -1.2, 4),
    ("2014-01-17", 0.9, 5),
    ("2014-02-15", -1.2, 5),
]


df_a = make_panel_data("a", anchors_a, events_a, seed=1)
df_b = make_panel_data("b", anchors_b, events_b, seed=2)

df = pd.concat([df_a, df_b], ignore_index=True)

# df 即为后续绘图使用的虚拟数据
print(df.head())
```

## 三、完整 Python 绘图代码

```python
import pandas as pd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
matplotlib.rcParams["svg.fonttype"] = "none"
import matplotlib.dates as mdates
from matplotlib.lines import Line2D
from matplotlib.ticker import MultipleLocator


# ======================
# 1. 基础风格参数
# ======================
plt.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Arial", "DejaVu Sans"],
    "axes.linewidth": 1.0,
    "axes.edgecolor": "black",
    "xtick.direction": "out",
    "ytick.direction": "out",
    "xtick.major.width": 0.9,
    "ytick.major.width": 0.9,
    "xtick.major.size": 3,
    "ytick.major.size": 3,
    "savefig.dpi": 300,
})


colors = {
    "PFOA_total_10*loading": "#ff1b1b",
    "PFOA_total_5*loading": "#7fc348",
    "PFOA_total_ 2*loading": "#ff 9 f 1 a",
    "PFOA_total": "#1 c 8 fd 2",
    "PNEC": "#d 9 d 9 d 9",
}

line_order = [
    "PFOA_total_ 10*loading",
    "PFOA_total_ 5*loading",
    "PFOA_total_ 2*loading",
    "PFOA_total",
]

panel_settings = {
    "a": {
        "ylim": (0, 160),
        "yticks": range(0, 161, 20),
    },
    "b": {
        "ylim": (0, 140),
        "yticks": range(0, 141, 20),
    }
}


# ======================
# 2. 创建画布
# ======================
fig, axes = plt.subplots(
    nrows=2,
    ncols=1,
    figsize=(5.99, 9.35),
    dpi=100,
    sharex=False
)

plt.subplots_adjust(
    left=0.13,
    right=0.975,
    top=0.985,
    bottom=0.165,
    hspace=0.255
)


# ======================
# 3. 绘制两个子图
# ======================
for ax, panel in zip(axes, ["a", "b"]):
    sub = df[df["panel"] == panel].copy()

    # PNEC 水平阈值线
    ax.axhline(
        y=50,
        color=colors["PNEC"],
        lw=0.9,
        zorder=0
    )

    # 情景折线
    for col in line_order:
        ax.plot(
            sub["date"],
            sub[col],
            color=colors[col],
            lw=0.85,
            solid_capstyle="round",
            label=col
        )

    # 坐标轴范围
    ax.set_xlim(pd.Timestamp("2013-03-01"), pd.Timestamp("2014-04-05"))
    ax.set_ylim(*panel_settings[panel]["ylim"])
    ax.set_yticks(panel_settings[panel]["yticks"])

    # 轴标签
    ax.set_ylabel(r"Concentration (ng L$^{-1}$)", fontsize=11)

    # 日期刻度
    ax.xaxis.set_major_locator(mdates.MonthLocator(interval=1))
    ax.xaxis.set_major_formatter(mdates.DateFormatter("%Y-%m-%d"))

    for tick in ax.get_xticklabels():
        tick.set_rotation(50)
        tick.set_ha("right")
        tick.set_fontsize(10)

    for tick in ax.get_yticklabels():
        tick.set_fontsize(10)

    # 浅灰水平参考线，模拟原图背景中的淡网格感
    ax.grid(
        axis="y",
        color="#e 6 e 6 e 6",
        linewidth=0.6,
        alpha=0.55
    )

    # 保留黑色边框
    for spine in ax.spines.values():
        spine.set_visible(True)
        spine.set_linewidth(1.0)
        spine.set_color("black")

    # 子图标签 a / b
    ax.text(
        -0.13,
        1.02,
        panel,
        transform=ax.transAxes,
        fontsize=12,
        fontweight="bold",
        va="top",
        ha="left"
    )


# ======================
# 4. 底部图例
# ======================
dummy = Line 2 D([0], [0], color="none", lw=0, label="")

legend_handles = [
    Line 2 D([0], [0], color=colors["PFOA_total_ 10*loading"], lw=1.0, label="PFOA_total_ 10*loading"),
    Line 2 D([0], [0], color=colors["PFOA_total_ 5*loading"], lw=1.0, label="PFOA_total_ 5*loading"),
    dummy,
    Line 2 D([0], [0], color=colors["PFOA_total_ 2*loading"], lw=1.0, label="PFOA_total_ 2*loading"),
    Line 2 D([0], [0], color=colors["PFOA_total"], lw=1.0, label="PFOA_total"),
    Line 2 D([0], [0], color=colors["PNEC"], lw=1.0, label="PNEC"),
]

leg = fig.legend(
    handles=legend_handles,
    loc="lower center",
    bbox_to_anchor=(0.5, 0.035),
    ncol=3,
    frameon=True,
    fancybox=False,
    edgecolor="black",
    framealpha=1,
    fontsize=9,
    handlelength=1.35,
    handletextpad=0.45,
    columnspacing=1.15,
    borderpad=0.65
)

leg.get_frame().set_linewidth(0.9)


# ======================
# 5. 保存与显示
# ======================
plt.savefig(
    "reproduced_pfoa_timeseries.png",
    dpi=300,
    bbox_inches="tight"
)

plt.close(fig)
```
