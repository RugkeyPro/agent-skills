一、图像类型识别与结构拆解

这是一张**双面板散点关系图 / 环境水文统计散点图**。

左图 **a**：

- 图类型：对数横轴散点图 + 连续变量颜色映射
    
- 横轴：`Discharge (m³/s)`，对数坐标
    
- 纵轴：`All retention proportion (%)`
    
- 颜色变量：`River length (km)`
    
- 色标：橙色表示短河流，蓝色表示长河流
    
- 点型：小圆点，半透明，灰色边框
    

右图 **b**：

- 图类型：多类别空心散点图 + 幂函数拟合曲线
    
- 横轴：`River velocity (m/s)`
    
- 纵轴：`Percentage (%)`
    
- 分组：`Fibers`、`Fragments`、`All`
    
- 点型：空心圆
    
- 拟合线：黑色虚线，幂函数趋势
    
- 注释：`y=0.31x^-0.30`，`R²=0.22, p<0.05`

## 二、虚拟数据生成代码

```python
import numpy as np
import pandas as pd


def generate_panel_a_data(seed=42, n=185):
    """
    生成 panel a 的虚拟数据：
    Discharge - All retention proportion - River length
    """
    rng = np.random.default_rng(seed)

    # 流量：对数分布，范围约 10^0 到 10^5
    discharge = 10 ** rng.uniform(0.2, 4.6, n)

    # 河流长度：与流量弱正相关，同时保留大量短河流
    river_length = 10 ** (
        0.2
        + 0.78 * np.log10(discharge)
        + rng.normal(0, 0.65, n)
    )
    river_length = np.clip(river_length, 1, 14500)

    # 滞留比例：整体随流量增大而下降，但具有较强离散性
    base = 95 * np.exp(-0.36 * np.log10(discharge))
    noise = rng.normal(0, 24, n)
    retention = base + noise

    # 人为加入部分高滞留点和低滞留点，使分布接近原图
    high_idx = rng.choice(n, size=28, replace=False)
    retention[high_idx] += rng.uniform(25, 70, len(high_idx))

    low_idx = rng.choice(n, size=45, replace=False)
    retention[low_idx] -= rng.uniform(20, 45, len(low_idx))

    retention = np.clip(retention, 0, 100)

    df = pd.DataFrame({
        "discharge": discharge,
        "retention": retention,
        "river_length": river_length
    })

    return df


def generate_panel_b_data(seed=7, n_each=125):
    """
    生成 panel b 的虚拟数据：
    River velocity - Percentage - Type
    """
    rng = np.random.default_rng(seed)

    records = []

    groups = {
        "Fibers": {
            "color": "#6aaa1f",
            "scale": 34,
            "noise": 19
        },
        "Fragments": {
            "color": "#ff1f1f",
            "scale": 26,
            "noise": 17
        },
        "All": {
            "color": "#5aa9ff",
            "scale": 30,
            "noise": 20
        }
    }

    for group, params in groups.items():
        # 大多数速度集中在 0-15，少量延伸到 80
        v1 = rng.gamma(shape=1.25, scale=4.2, size=int(n_each * 0.85))
        v2 = rng.uniform(15, 80, size=n_each - len(v1))
        velocity = np.concatenate([v1, v2])
        rng.shuffle(velocity)
        velocity = np.clip(velocity, 0.15, 82)

        # 百分比随速度增加整体下降，低速处离散性强
        trend = params["scale"] * (velocity + 0.4) ** (-0.42)
        noise = rng.normal(0, params["noise"], n_each) * np.exp(-0.035 * velocity)

        percentage = trend + noise + rng.exponential(4.0, n_each)
        percentage = np.clip(percentage, 0, 100)

        # 增加低速高百分比点
        low_v = np.where(velocity < 5)[0]
        if len(low_v) > 0:
            chosen = rng.choice(low_v, size=min(18, len(low_v)), replace=False)
            percentage[chosen] += rng.uniform(25, 70, len(chosen))
            percentage = np.clip(percentage, 0, 100)

        for x, y in zip(velocity, percentage):
            records.append({
                "velocity": x,
                "percentage": y,
                "type": group
            })

    df = pd.DataFrame(records)
    return df


if __name__ == "__main__":
    df_a = generate_panel_a_data()
    df_b = generate_panel_b_data()

    df_a.to_csv("panel_a_retention_data.csv", index=False)
    df_b.to_csv("panel_b_velocity_data.csv", index=False)

    print(df_a.head())
    print(df_b.head())
```
