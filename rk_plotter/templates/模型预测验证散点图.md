一、图像类型识别与结构拆解

这是一张**双面板模型预测验证散点图 / Parity plot / Predicted vs Real scatter plot**。

```text
a) XGBoost
b) Transformer
```

每个子图均包含：

- 横轴：`Real removal (%)`
    
- 纵轴：`Pred removal (%)`
    
- 红色参考线：`y = x`
    
- 蓝色散点：Training data
    
- 红色散点：Testing data
    
- 左上角文字：训练集与测试集误差指标，包括 RMSE 和 MAE
    
- 子图 b 中含图例
    
- 整体风格：粗边框、大字号、论文模型验证图风格

## 二、单独的虚拟数据生成代码

```python
import numpy as np
import pandas as pd


def make_model_validation_data(seed=42):
    """
    生成用于复现图像的虚拟数据。
    字段：
    model      : XGBoost / Transformer
    dataset    : Training data / Testing data
    real       : Real removal (%)
    pred       : Pred removal (%)
    """

    rng = np.random.default_rng(seed)
    records = []

    # =========================
    # a) XGBoost
    # 训练集：高度贴近 y=x
    # 测试集：主要集中在高去除率区域，略有离散
    # =========================
    n_train_xgb = 120
    real_train = rng.uniform(55, 100, n_train_xgb)
    pred_train = real_train + rng.normal(0, 0.9, n_train_xgb)
    pred_train = np.clip(pred_train, 0, 105)

    for x, y in zip(real_train, pred_train):
        records.append(["XGBoost", "Training data", x, y])

    n_test_xgb = 75
    real_test = rng.uniform(68, 100, n_test_xgb)
    pred_test = 93 + 0.20 * (real_test - 80) + rng.normal(0, 4.0, n_test_xgb)
    pred_test = np.clip(pred_test, 80, 103)

    for x, y in zip(real_test, pred_test):
        records.append(["XGBoost", "Testing data", x, y])

    # =========================
    # b) Transformer
    # 训练集、测试集均较离散，但仍围绕 y=x 附近
    # =========================
    n_train_tf = 145
    real_train = rng.uniform(55, 100, n_train_tf)
    pred_train = 0.72 * real_train + 26 + rng.normal(0, 8.0, n_train_tf)
    pred_train = np.clip(pred_train, 45, 103)

    for x, y in zip(real_train, pred_train):
        records.append(["Transformer", "Training data", x, y])

    n_test_tf = 95
    real_test = rng.uniform(70, 100, n_test_tf)
    pred_test = 0.80 * real_test + 18 + rng.normal(0, 5.5, n_test_tf)
    pred_test = np.clip(pred_test, 55, 103)

    for x, y in zip(real_test, pred_test):
        records.append(["Transformer", "Testing data", x, y])

    df = pd.DataFrame(
        records,
        columns=["model", "dataset", "real", "pred"]
    )

    return df


df = make_model_validation_data()
print(df.head())
```

## 三、完整 Python 绘图代码

```python
import numpy as np
import pandas as pd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
matplotlib.rcParams["svg.fonttype"] = "none"
from matplotlib.lines import Line2D


# ============================================================
# 1. 虚拟数据生成
# ============================================================

def make_model_validation_data(seed=42):
    rng = np.random.default_rng(seed)
    records = []

    # XGBoost training
    n_train_xgb = 120
    real_train = rng.uniform(55, 100, n_train_xgb)
    pred_train = real_train + rng.normal(0, 0.9, n_train_xgb)
    pred_train = np.clip(pred_train, 0, 105)

    for x, y in zip(real_train, pred_train):
        records.append(["XGBoost", "Training data", x, y])

    # XGBoost testing
    n_test_xgb = 75
    real_test = rng.uniform(68, 100, n_test_xgb)
    pred_test = 93 + 0.20 * (real_test - 80) + rng.normal(0, 4.0, n_test_xgb)
    pred_test = np.clip(pred_test, 80, 103)

    for x, y in zip(real_test, pred_test):
        records.append(["XGBoost", "Testing data", x, y])

    # Transformer training
    n_train_tf = 145
    real_train = rng.uniform(55, 100, n_train_tf)
    pred_train = 0.72 * real_train + 26 + rng.normal(0, 8.0, n_train_tf)
    pred_train = np.clip(pred_train, 45, 103)

    for x, y in zip(real_train, pred_train):
        records.append(["Transformer", "Training data", x, y])

    # Transformer testing
    n_test_tf = 95
    real_test = rng.uniform(70, 100, n_test_tf)
    pred_test = 0.80 * real_test + 18 + rng.normal(0, 5.5, n_test_tf)
    pred_test = np.clip(pred_test, 55, 103)

    for x, y in zip(real_test, pred_test):
        records.append(["Transformer", "Testing data", x, y])

    return pd.DataFrame(
        records,
        columns=["model", "dataset", "real", "pred"]
    )


df = make_model_validation_data()


# ============================================================
# 2. 样式参数
# ============================================================

plt.rcParams.update({
    "font.family": "sans-serif",
    "font.sans-serif": ["Arial", "DejaVu Sans"],
    "axes.linewidth": 2.1,
    "xtick.direction": "in",
    "ytick.direction": "in",
    "xtick.major.size": 5,
    "ytick.major.size": 5,
    "xtick.major.width": 1.5,
    "ytick.major.width": 1.5,
    "savefig.dpi": 300,
})

colors = {
    "Training data": "#8794f2",
    "Testing data": "#e97865",
}

metrics_text = {
    "XGBoost": "Training RMSE: 0.7%\nTraining MAE: 0.5%\nRMSE: 8.4%\nMAE: 6.4%",
    "Transformer": "Training RMSE: 5.1%\nTraining MAE: 3.3%\nRMSE: 5.9%\nMAE: 4.5%",
}

titles = {
    "XGBoost": "a) XGBoost",
    "Transformer": "b) Transformer",
}


# ============================================================
# 3. 绘图
# ============================================================

fig, axes = plt.subplots(
    nrows=1,
    ncols=2,
    figsize=(8.9, 4.0),
    dpi=120
)

for ax, model in zip(axes, ["XGBoost", "Transformer"]):
    sub = df[df["model"] == model]

    # 1:1 参考线
    ax.plot(
        [0, 100],
        [0, 100],
        color="red",
        lw=1.25,
        zorder=1
    )

    # 训练集与测试集散点
    for dataset in ["Training data", "Testing data"]:
        ss = sub[sub["dataset"] == dataset]

        ax.scatter(
            ss["real"],
            ss["pred"],
            s=28,
            color=colors[dataset],
            edgecolor=colors[dataset],
            alpha=0.72,
            linewidth=0.5,
            zorder=3
        )

    # 坐标范围与刻度
    ax.set_xlim(-10, 120)
    ax.set_ylim(-10, 120)

    ax.set_xticks([0, 25, 50, 75, 100])
    ax.set_yticks([0, 25, 50, 75, 100])

    ax.tick_params(labelsize=16)

    ax.set_xlabel("Real removal (%)", fontsize=24, labelpad=8)
    ax.set_ylabel("Pred removal (%)", fontsize=22, labelpad=8)

    # 标题
    ax.set_title(
        titles[model],
        fontsize=21,
        pad=10
    )

    # 误差文本
    ax.text(
        0.035,
        0.955,
        metrics_text[model],
        transform=ax.transAxes,
        ha="left",
        va="top",
        fontsize=13.5,
        linespacing=1.15
    )

    # 粗黑边框
    for spine in ax.spines.values():
        spine.set_linewidth(2.1)
        spine.set_color("black")


# ============================================================
# 4. 图例，仅放在 Transformer 子图内
# ============================================================

legend_handles = [
    Line2D(
        [0], [0],
        marker="o",
        linestyle="none",
        markersize=9,
        markerfacecolor=colors["Training data"],
        markeredgecolor=colors["Training data"],
        alpha=0.85,
        label="Training data"
    ),
    Line2D(
        [0], [0],
        marker="o",
        linestyle="none",
        markersize=9,
        markerfacecolor=colors["Testing data"],
        markeredgecolor=colors["Testing data"],
        alpha=0.85,
        label="Testing data"
    )
]

axes[1].legend(
    handles=legend_handles,
    loc="lower right",
    frameon=False,
    fontsize=16,
    handlelength=0.8,
    handletextpad=0.5,
    borderpad=0.2
)


# ============================================================
# 5. 布局与保存
# ============================================================

plt.subplots_adjust(
    left=0.08,
    right=0.985,
    bottom=0.22,
    top=0.86,
    wspace=0.36
)

plt.savefig(
    "reproduced_model_validation_scatter.png",
    dpi=300,
    bbox_inches="tight",
    pad_inches=0.04
)

plt.close(fig)
```

## 四、可调整参数说明

控制 XGBoost 训练集贴近程度：

```python
pred_train = real_train + rng.normal(0, 0.9, n_train_xgb)
```

数值越小，点越贴近红色 `y=x` 线。

控制 Transformer 离散程度：

```python
pred_train = 0.72 * real_train + 26 + rng.normal(0, 8.0, n_train_tf)
```

增大 `8.0` 会让散点更分散。

控制散点大小与透明度：

```python
s=28
alpha=0.72
```

控制坐标轴粗细：

```python
"axes.linewidth": 2.1
```
